        public void SpeechRecognizedFunction(object sender, SpeechRecognizedEventArgs e) {
            //if (e.Result.Confidence < 0.5) {
            //    return;
            //}
            RecognitionResult WordAux;
            WordAux = e.Result;

            string strWord = WordAux.Text.ToLower();

            switch(strWord) {
                case "samanta":
                    if(SamanthaVoice_AcceptCommand == true) {
                        return;
                    }
                    //  Disponible para aceptar comandos / Begin capture CommandArmed
                    SamanthaVoice_AcceptCommand = true;
                    //  Tiempo de espera para aceptar todos los comandos
                    _CountRemaining = ticksRemaining;

                    //  Variable aleatoria

                    String[] samVoiceSplit = this.samanthaVoice.OrderYou.Split(',');
                    //  La variable random solo acepta valores que van de cero hasta el numero de elementos de "yesSr"
                    SpeechText(samVoiceSplit[_rnd.Next(0, samVoiceSplit.Length)]);
                    EventSamanthaActivated(sender, e);
                    break;
                default:
                    // Esta disponible para aceptar comandos ?
                    if(SamanthaVoice_AcceptCommand == true) {
                        /*
                        ObjectWord myObjetc = new ObjectWord();
                        myObjetc.TextToSearch = strWord;
                        myObjetc.EventCompleted += new EventHandler(ProcessContinue);
                        myObjetc.Initialize();
                        myObjetc.Run();
                        */

                        //  Concatena la palabra y le aumenta una coma al final
                        strBuildingInstruction = strBuildingInstruction + strWord + ",";

                        //  Tiempo de espera para aceptar todos los comandos
                        _CountRemaining = ticksRemaining;
                        EventTimeRemaining(_CountRemaining, e);
                        EventWordRecognizedAccepted(strWord, e);
                    }
                    EventWordRecognized(strWord, e);
                    break;
            }
            string alternates = string.Empty;
            foreach(var alt in e.Result.Alternates) {
                var altConf = alt.Confidence;
                string words = string.Empty;
                if(alt.Words.Count > 1) {
                    foreach(var word in alt.Words) {
                        words = $"{words}_{word.Text}:{word.Confidence}";
                    }
                }
                string straux = alt.Words.Count > 1 ? $"[_WORDS:{words}] ---" : string.Empty;
                straux = string.Empty;
                string myWORD = alt.Confidence > 0.65 ? $"*{alt.Text.ToUpper()}" : alt.Text.ToLower();
                alternates = $"{alternates}  {myWORD}:{alt.Confidence}{straux}";
            }
            _ = WriteToFile("SpeechRecognized", $"{alternates.Trim()}");
        }

        private void timerRemaining_Tick(object sender, System.Timers.ElapsedEventArgs e) {

                }
            }

        void WordConcatFunction(string strSplit) {
            string[] listWord = strSplit.Split(',');
            foreach(string item in listWord) {
                WordsConcat.Add(item);
            }
        }

        void ReservedWordListFunction(string strSplit) {
            string[] listWord = strSplit.Split(',');
            foreach(string item in listWord) {
                reservedWordlist.Add(item);
            }
        }

        void RecognizedWordFunction(RecognizedWordEn recognizedWord) {
            string wordClass = recognizedWord.WordClass;
            string codeWord = recognizedWord.Code;
            string[] listWord = recognizedWord.RelatedWords.Split(',');
            foreach(string item in listWord) {
                ValidWordEn model = new ValidWordEn();
                model.WordClass = wordClass;
                model.CodeWord = codeWord;
                model.ValueWord = item;
                modelWordList.Add(model);
            }
        }

        public void GetInstruction(string strToSplit) {
            //  We create the object (variable) "cSamanthaAnalizer" of class "Analizer"
            //  -- Creamos el objeto (variable) "cSamanthaAnalizer" de la clase "Analizer"
            Recognize cSamanthaAnalizer = new Recognize(recognizedInstructionList);

            //  We create the object (variable) "cCommandVoice" of class "CommandVoice"
            //  It determines the nature of each word (type of noun): Verb, Adjective, subject, pronoun, article, question, numerator, etc.
            //  And save it in the structure (object) "cCommandVoice"
            //  -- Creamos el objeto (variable) "cCommandVoice" de la clase "CommandVoice"
            //  -- Determina la naturaleza de cada palabra (tipo de sustantivo): Verbo, Adjetivo, sujeto, pronombre, articulo, pregunta, numerador, etc
            //  -- y la guardamos en la estructura (objeto) "cCommandVoice"
            string result = string.Empty;
            List<RecognizedInstructionEn> listRecognizedInstruction =
                cSamanthaAnalizer.GetRecognizedInstructions(
                    strToSplit,
                    reservedWordlist,
                    modelWordList,
                    out result
                );
            //  If there were no mistakes then ...
            //  -- Si no hubo Errores entonces ...
            switch (result) {
                case "OK":
                    //  We created the event indicating that we found a solution
                    EventSolutionFound(listRecognizedInstruction, null);
                    break;
                case "NOINSTRUCTION":
                    SpeechText(samanthaVoice.ThereIsNoInstruction);
                    break;
                case "INSTRUCTIONNOTFOUND":
                    SpeechText(samanthaVoice.InstructionNotFound);
                    break;
                case "EXCEPTION":
                    SpeechText(samanthaVoice.AnExceptionOcurred);
                    break;
                default:
                    break;
            }
        }

        private async Task WriteToFile(string fileName, string strInfo) {
            try {
                string nameDirectoryMonth = DateTime.Now.ToString("yyyy-MM");
                if(!Directory.Exists(nameDirectoryMonth))
                    Directory.CreateDirectory(nameDirectoryMonth);
                string nameDirectoryDay = DateTime.Now.ToString("yyyy-MM-dd");
                if(!Directory.Exists($"{nameDirectoryMonth}\\{nameDirectoryDay}"))
                    Directory.CreateDirectory($"{nameDirectoryMonth}\\{nameDirectoryDay}");
                //
                string nameFileComplement = DateTime.Now.ToString("yyyyMMddHH");
                string sNombreTXTSession = $"{nameDirectoryMonth}\\{nameDirectoryDay}\\samantha_{fileName}_{nameFileComplement}.txt";
                StringBuilder sb = new StringBuilder();

                sb.AppendLine($"{DateTime.Now.ToString("yyy-MM-dd HH:mm:ss.ffff ")} {strInfo}");
                using(StreamWriter archivo = new StreamWriter(sNombreTXTSession, true)) {
                    archivo.Write(sb.ToString());
                }
            } catch(Exception ex) {
                Console.WriteLine("Exception: " + ex.Message);
            } finally {
                Console.WriteLine("Executing finally block.");
            }
            await Task.CompletedTask;
        }

public class ObjectWord {

    public int id { get; set; }
    public string TextToSearch {  get; set; }
    public BackgroundWorker BWWordProcess { get; set; }
    public event EventHandler EventCompleted;

    public ObjectWord() {

        BWWordProcess = new BackgroundWorker();
        BWWordProcess.WorkerSupportsCancellation = true;
    }

    public void Initialize() {
        BWWordProcess.DoWork += new DoWorkEventHandler(BWWordProcess_DoWork);
        BWWordProcess.RunWorkerCompleted += new RunWorkerCompletedEventHandler(BWWordProcess_Compoleto);
    }

    public void Run() {
        BWWordProcess.RunWorkerAsync();
    }

    public void BWWordProcess_DoWork(object sender, DoWorkEventArgs e) {
        TextToSearch = TextToSearch + DateTime.Now.ToString();
    }

    public void BWWordProcess_Compoleto(object sender, RunWorkerCompletedEventArgs e) {
        EventCompleted(TextToSearch, null);
    }
}
